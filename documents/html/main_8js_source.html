85async function downloadMaterial() {
86 if (!currentSelectedMaterial || !polyHavenAPI) return;
87
88 const resolution = document.getElementById('materialResolution').value;
89 const materialName = currentSelectedMaterial.name.replace(/[^a-z0-9]/gi, '_').toLowerCase();
90
91 // Show loading state
92 const downloadBtn = document.getElementById('downloadMaterial');
93 const originalText = downloadBtn.innerHTML;
94 downloadBtn.innerHTML = '<i class="bi bi-arrow-clockwise spin me-2"></i>Preparing download...';
95 downloadBtn.disabled = true;
96
97 try {
98 // Create the MaterialX package using the API class
99 const zipBlob = await polyHavenAPI.createMaterialXPackage(currentSelectedMaterial, resolution);
100
101 // Trigger download
102 const url = URL.createObjectURL(zipBlob);
103 const a = document.createElement('a');
104 a.href = url;
105 a.download = `${materialName}_${resolution}_materialx.zip`;
106 document.body.appendChild(a);
107 a.click();
108
109 // Clean up
110 setTimeout(() => {
111 document.body.removeChild(a);
112 URL.revokeObjectURL(url);
113 }, 100);
114
115 } catch (error) {
116 console.error('Error creating MaterialX package:', error);
117 alert(`Failed to prepare download: ${error.message}`);
118 } finally {
119 // Restore button state
120 downloadBtn.innerHTML = originalText;
121 downloadBtn.disabled = false;
122 }
123}

174// Filter materials based on search and category filters
175function filterMaterials() {
176 const searchTerm = searchInput.value.toLowerCase();
177 const selectedCategory = categoryFilter.value;
178
179 const filtered = allMaterials.filter(material => {
180 // Filter by search term
181 const matchesSearch = material.name.toLowerCase().includes(searchTerm) ||
182 material.description.toLowerCase().includes(searchTerm) ||
183 material.tags.some(tag => tag.toLowerCase().includes(searchTerm));
184
185 // Filter by category
186 const matchesCategory = !selectedCategory || material.categories.includes(selectedCategory);
187
188 return matchesSearch && matchesCategory;
189 });
190
191 displayMaterials(filtered);
192}
