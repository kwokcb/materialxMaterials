48 def fetch_materialx_assets(self, max_items=1, download_id=None, download_type=None):
49 '''
50 Fetch MaterialX assets from PolyHaven API and filter them by resolution.
51
52 @param max_items Maximum number of assets to fetch. If None, fetch all.
53 @param download_id If set, only fetch asset with this ID.
54 @param download_type Type of asset to download (e.g. 'mtlx', 'blend', 'gltf'). Default is None which means to fetch MaterialX
55 @return Tuple (materialx_assets, all_assets, filtered_polyhaven_assets):
56 - materialx_assets: Dictionary of MaterialX assets with URLs and texture files.
57 - all_assets: All assets returned by PolyHaven API.
58 - filtered_polyhaven_assets: Filtered assets containing only MaterialX files.
59 '''
60 parameters = {
61 "type": "textures"
62 }
63
64 resp = requests.get(self.ASSET_API, headers=self.HEADERS, params=parameters)
65 resp.raise_for_status()
66 all_assets = resp.json()
67
68 materialx_assets = {}
69 gltf_assets = {}
70 blender_assets = {}
71 filtered_polyhaven_assets = {}
72
73 # Look for 1K, 2K , 4K, and 8K versions
74 resolutions = {
75 "1k": None,
76 "2k": None,
77 "4k": None,
78 "8k": None
79 }
80
81 # Default to fetching MaterialX assets if no download type specified
82 if not download_type:
83 download_type = self.MTLX_KEY
84
85 item_count = 0
86 for id, data in all_assets.items():
87
88 if download_id and id != download_id:
89 #self.logger.info(f"Skipping asset id: '{id}' (not matching {download_id})")
90 continue
91
92 # Get the thumbnail
93 thumbnail_url = data.get("thumbnail_url")
94
95 resp = requests.get(f"{self.FILES_API}/{id}", headers=self.HEADERS)
96 resp.raise_for_status()
97 files_data = resp.json()
98 #json_string = json.dumps(files_data, indent=4)
99 #print(f"Files data for asset '{id}': {json_string}")
100
101 blend_files = files_data.get(self.BLEND_KEY, [])
102 if blend_files:
103 self.logger.info(f"Found Blender files for '{id}'")
104
105 for resolution_key in resolutions.keys():
106 res = blend_files.get(resolution_key, None)
107 if not res:
108 continue
109
110 n_k_mtlx = res.get(self.BLEND_KEY)
111 texture_struct = {}
112 if n_k_mtlx:
113 include_files = n_k_mtlx.get("include", {})
114 for path, data in include_files.items():
115 texture_url = data.get("url")
116 texture_struct[path] = texture_url
117 url = n_k_mtlx.get("url")
118 res_id = id + '_' + resolution_key
119 if url:
120 blender_assets[res_id] = {
121 "url": url,
122 "texture_files": texture_struct,
123 "thumbnail_url": thumbnail_url
124 }
125
126 if download_id == id and download_type == self.BLEND_KEY:
127 break
128
129 gltf_files = files_data.get(self.GLTF_KEY, [])
130 if gltf_files:
131 self.logger.info(f"Found glTF files for '{id}'")
132
133 for resolution_key in resolutions.keys():
134 res = gltf_files.get(resolution_key, None)
135 if not res:
136 continue
137
138 n_k_mtlx = res.get(self.GLTF_KEY)
139 texture_struct = {}
140 if n_k_mtlx:
141 include_files = n_k_mtlx.get("include", {})
142 for path, data in include_files.items():
143 texture_url = data.get("url")
144 texture_struct[path] = texture_url
145 url = n_k_mtlx.get("url")
146 res_id = id + '_' + resolution_key
147 if url:
148 gltf_assets[res_id] = {
149 "url": url,
150 "texture_files": texture_struct,
151 "thumbnail_url": thumbnail_url
152 }
153
154 # Halt if download_id is specified and matches the current asset ID and download type
155 if download_id == id and download_type == self.GLTF_KEY:
156 break
157
158
159 # Remove all keys other than "mtlx"
160 #files_data = {k: v for k, v in files_data.items() if k == "mtlx"}
161 mtlx_files = files_data.get(self.MTLX_KEY, [])
162 if mtlx_files:
163
164 self.logger.info(f"Found MaterialX data for '{id}'")
165
166 for resolution_key in resolutions.keys():
167 res = mtlx_files.get(resolution_key, None)
168 if not res:
169 continue
170
171 n_k_mtlx = res.get(self.MTLX_KEY)
172 texture_struct = {}
173 if n_k_mtlx:
174 include_files = n_k_mtlx.get("include", {})
175 for path, data in include_files.items():
176 texture_url = data.get("url")
177 texture_struct[path] = texture_url
178 mtlx_url = n_k_mtlx.get("url")
179 res_id = id + '_' + resolution_key
180 if mtlx_url:
181 materialx_assets[res_id] = {
182 "url": mtlx_url,
183 "texture_files": texture_struct,
184 "thumbnail_url": thumbnail_url
185 }
186
187 filtered_polyhaven_assets[id] = files_data
188
189 # Halt if download_id is specified and matches the current asset ID and download type
190 if download_id == id and download_type == self.MTLX_KEY:
191 break
192
193 # Halt if max_items is specified and reached
194 if not download_id and max_items:
195 item_count += 1
196 if item_count >= max_items:
197 break
198
199 #if "materialx" in formats:
200 # materialx_assets[slug] = formats["materialx"]
201
202 return materialx_assets, all_assets, filtered_polyhaven_assets, blender_assets, gltf_assets

204 def download_gltf_asset(self, asset_list):
205 '''
206 Download glTF asset from PolyHaven.
207
208 @param asset_list Dictionary of glTF assets with URLs and texture files.
209 @return Tuple (id, gltf_ascii, texture_binaries):
210 - id: ID of the downloaded asset.
211 - gltf_ascii: The glTF file as ASCII string.
212 - texture_binaries: List of tuples (path, binary content) for textures and thumbnails
213 '''
214 for id, asset in asset_list.items():
215 url = asset.get("url")
216 if not url:
217 self.logger.info(f"No glTF URL found for '{id}'")
218 continue
219 if not url.endswith(".gltf") and not url.endswith(".glb"):
220 self.logger.info(f"Invalid glTF URL for '{id}': {url}")
221 continue
222
223 resp = requests.get(url, headers=self.HEADERS)
224 resp.raise_for_status()
225 if url.endswith(".glb"):
226 self.logger.info(f"Downloaded glTF binary file {url}, size: {len(resp.content)} bytes")
227 gltf_content = resp.content
228 elif url.endswith(".gltf"):
229 gltf_content = resp.text
230 self.logger.info(f"Download glTF file {url}, length: {len(gltf_content)} characters")
231
232 texture_binaries = []
233 for path, texture_url in asset.get("texture_files", {}).items():
234 self.logger.info(f"Download texture from {texture_url} ...")
235 texture_resp = requests.get(texture_url, headers=self.HEADERS)
236 texture_resp.raise_for_status()
237 texture_content = texture_resp.content
238 texture_binaries.append((path, texture_content))
239
240 thumbnail_url = asset.get("thumbnail_url")
241 if thumbnail_url:
242 self.logger.info(f"Download thumbnail from {thumbnail_url} ...")
243 thumbnail_resp = requests.get(thumbnail_url, headers=self.HEADERS)
244 thumbnail_resp.raise_for_status()
245
246 clean_url = thumbnail_url
247 # Strip any ? or # from the URL
248 clean_url = clean_url.split('?')[0].split('#')[0]
249 clean_url = clean_url.split('/')[-1] # Get the last part of the URL
250 extension = Path(clean_url).suffix.lower()
251 texture_binaries.append((f"{id}_thumbnail{extension}", thumbnail_resp.content))
252 return id, gltf_content, texture_binaries

297 def download_mtlx_asset(self, asset_list, convert_exr_to_png=True):
298 '''
299 Download MaterialX asset and its textures from PolyHaven.
300
301 @param asset_list Dictionary of MaterialX assets with URLs and texture files.
302 @param convert_exr_to_png If True, attempt to use PNG images instead of EXR if available other attempt
303 to convert using OpenImageIO if installed. Default is to use PNG if possible, and convert EXR to PNG if OpenImageIO is available.
304 @return Tuple (id, mtlx_string, texture_binaries):
305 - id: ID of the downloaded asset.
306 - mtlx_string: The MaterialX document as a string.
307 - texture_binaries: List of tuples (path, binary content) for textures and thumbnails.
308 '''
309 for id, asset in asset_list.items():
310 url = asset.get("url")
311 if not url:
312 self.logger.info(f"No MaterialX URL found for '{id}'")
313 continue
314
315 resp = requests.get(url, headers=self.HEADERS)
316 resp.raise_for_status()
317 mtlx_string = resp.text
318 self.logger.info(f"Download MaterialX document {url}, length: {len(mtlx_string)} characters")
319
320 texture_binaries = []
321 download_texture_names = []
322 for path, texture_url in asset.get("texture_files", {}).items():
323 # Get texture files
324 ext = Path(path).suffix.lower()
325 texture_content = None
326
327 if ext == ".exr" and convert_exr_to_png:
328 # Replace /exr and .exr with /png and .png in the URL to check if a PNG version is available
329 old_texture_url = texture_url
330 texture_url = texture_url.replace("/exr/", "/png/").replace(".exr", ".png")
331 texture_resp = requests.get(texture_url, headers=self.HEADERS)
332 texture_resp.raise_for_status()
333 texture_content = texture_resp.content
334 if texture_content:
335 self.logger.info(f"Download {texture_url} instead of {old_texture_url} SUCCESSFUL")
336 ext = ".png"
337 # Update extension in the path to .png
338 old_path = path
339 path = str(Path(path).with_suffix(ext))
340 self.logger.info(f"- Updating texture path from {old_path} to {path}")
341
342 else:
343 self.logger.info(f"Download {texture_url} instead of {old_texture_url} FAILED")
344
345 if not texture_content:
346 self.logger.info(f"Download texture from {texture_url} ...")
347 texture_resp = requests.get(texture_url, headers=self.HEADERS)
348 texture_resp.raise_for_status()
349 texture_content = texture_resp.content
350
351 name = Path(path).stem
352
353 if ext == ".exr":
354 if HAVE_OPENIMAGEIO and convert_exr_to_png:
355 self.logger.info(f"Converting EXR to PNG for texture: {path}")
356 # Write EXR bytes to a temporary file
357 with tempfile.NamedTemporaryFile(suffix=".exr", delete=False) as tmp_exr:
358 tmp_exr.write(texture_content)
359 tmp_exr_path = tmp_exr.name
360 try:
361 inbuf = oiio.ImageInput.open(tmp_exr_path)
362 if inbuf:
363 spec = inbuf.spec()
364 pixels = inbuf.read_image(format=oiio.UINT8)
365 inbuf.close()
366 # Write PNG to another temporary file
367 with tempfile.NamedTemporaryFile(suffix=".png", delete=False) as tmp_png:
368 outbuf = oiio.ImageOutput.create(tmp_png.name)
369 if outbuf:
370 outbuf.open(tmp_png.name, spec)
371 outbuf.write_image(pixels)
372 outbuf.close()
373 tmp_png.seek(0)
374 png_bytes = tmp_png.read()
375 png_name = f"{name}.png"
376 texture_binaries.append((png_name, png_bytes))
377
378 continue # Skip adding the original EXR
379 else:
380 self.logger.info("Failed to read EXR with OpenImageIO")
381 finally:
382 os.remove(tmp_exr_path)
383 if 'tmp_png' in locals():
384 os.remove(tmp_png.name)
385 self.logger.info(f" WARNING: EXR file present which may not be supported by MaterialX texture loader: {path}")
386
387 # Get file name from path
388 download_texture_name = path.split('/')[-1]
389 download_texture_names.append(download_texture_name)
390
391 texture_binaries.append((path, texture_content))
392
393 thumbnail_url = asset.get("thumbnail_url")
394 if thumbnail_url:
395 self.logger.info(f"Download thumbnail from {thumbnail_url} ...")
396 thumbnail_resp = requests.get(thumbnail_url, headers=self.HEADERS)
397 thumbnail_resp.raise_for_status()
398
399 clean_url = thumbnail_url
400 # Strip any ? or # from the URL
401 clean_url = clean_url.split('?')[0].split('#')[0]
402 clean_url = clean_url.split('/')[-1] # Get the last part of the URL
403 extension = Path(clean_url).suffix.lower()
404 texture_binaries.append((f"{id}_thumbnail{extension}", thumbnail_resp.content))
405
406 # Replace .exr with .png in mtlx_string
407 for name in download_texture_names:
408 extension = Path(name).suffix.lower()
409 exr_name = name.replace(extension, ".exr")
410 # Replace exr_name with name in the mtlx_string
411 mtlx_string = mtlx_string.replace(exr_name, name)
412
413 before_mtlx_string = mtlx_string
414 mtlx_string = mtlx_string.replace(".exr", ".png")
415 if (before_mtlx_string != mtlx_string):
416 self.logger.info(f"Updated MaterialX string to reference PNG texture instead of EXR for {path}")
417 #self.logger.info(mtlx_string)
418
419 return id, mtlx_string, texture_binaries

421 def save_blender_with_textures(self, id, blend_binary, texture_binaries, data_folder, extract_zip=False):
422 '''
423 Save Blender file and texture binaries to a zip file.
424
425 @param id The ID of the Blender asset.
426 @param blend_binary The Blender file as binary content.
427 @param texture_binaries List of tuples (path, binary content) for textures and thumbnails.
428 @param data_folder Folder to save the zip file.
429 @param extract_zip If True, extract the zip file after saving zip.
430 @return None
431 '''
432 # Create a zip file with Blender file and textures
433 filename = f"{id}_blender.zip"
434 filename = Path(data_folder) / filename
435 with zipfile.ZipFile(filename, "w") as zipf:
436 # Write Blender file
437 zipf.writestr(f"{id}.blend", blend_binary)
438 # Write texture files
439 for path, content in texture_binaries:
440 zipf.writestr(path, content)
441 self.logger.info(f"Saved zip: {filename}")
442
443 # Save zip contents to folder
444 extract_folder = Path(data_folder) / f"{id}_blender"
445 if extract_zip:
446 with zipfile.ZipFile(filename, "r") as zipf:
447 zipf.extractall(extract_folder)
448 self.logger.info(f"Extracted zip contents to folder: {extract_folder}")
