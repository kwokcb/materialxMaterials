38 processMaterialsData(rawData) {
39 const materials = [];
40 const categories = new Set();
41
42 for (const [id, materialData] of Object.entries(rawData)) {
43 // Skip materials without required data
44 if (!materialData.name || !materialData.categories) continue;
45
46 // Process authors
47 let authorsList = 'Author(s): ';
48 const authors = materialData.authors || {};
49 const authorNames = Object.keys(authors);
50 authorsList += authorNames.join(', ');
51
52 let max_resolution = materialData.max_resolution
53 let maxresolutionString = '';
54 if (max_resolution) {
55 maxresolutionString = `${max_resolution[0]} x ${max_resolution[1]}`
56 }
57
58 const material = {
59 id,
60 name: materialData.name,
61 description: authorsList,
62 categories: materialData.categories,
63 tags: materialData.tags || [],
64 thumb_url: `https://cdn.polyhaven.com/asset_img/thumbs/${id}.png?width=512`,
65 maps: materialData.maxresolutionString || {}
66 };
67
68 materials.push(material);
69
70 // Collect categories
71 material.categories.forEach(cat => categories.add(cat));
72 }
73
74 return { materials, categories };
75 }

171 async createMaterialXPackage(material, resolution) {
172 try {
173 // Fetch MaterialX files data
174 const filesData = await this.fetchMaterialFiles(material.id);
175 const mtlxData = filesData.mtlx?.[resolution]?.mtlx;
176
177 if (!mtlxData) {
178 throw new Error(`No MaterialX files found for ${resolution} resolution`);
179 }
180
181 // Create ZIP file
182 const zip = new JSZip();
183
184 // 1. Download and add the main MaterialX file
185 const mtlxContent = await this.downloadMaterialXContent(mtlxData.url);
186 zip.file(`${material.id}.mtlx`, mtlxContent);
187
188 // 2. Download and add all included texture files
189 const textureFiles = mtlxData.include || {};
190 const texturePromises = Object.entries(textureFiles).map(async ([path, fileData]) => {
191 try {
192 const textureBlob = await this.downloadTexture(fileData.url);
193
194 // Maintain the folder structure from the include paths
195 const pathParts = path.split('/');
196 let currentFolder = zip;
197
198 // Handle nested folder structure
199 for (let i = 0; i < pathParts.length - 1; i++) {
200 const folderName = pathParts[i];
201 currentFolder = currentFolder.folder(folderName);
202 }
203
204 currentFolder.file(pathParts[pathParts.length - 1], textureBlob);
205
206 if (path.toLowerCase().endsWith('.exr')) {
207 console.warn(`EXR file present which may not be supported by MaterialX texture loader: ${path}`);
208 }
209 } catch (error) {
210 console.error(`Error downloading texture ${path}:`, error);
211 // Add placeholder file if download fails
212 zip.file(path, `Failed to download: ${fileData.url}`);
213 }
214 });
215
216 // 3. Download and add thumbnail
217 if (material.thumb_url) {
218 try {
219 const thumbBlob = await this.downloadThumbnail(material.thumb_url);
220 const thumbUrl = new URL(material.thumb_url);
221 const thumbPath = thumbUrl.pathname.split('/').pop();
222 const thumbExt = thumbPath.split('.').pop();
223 zip.file(`${material.id}_thumbnail.${thumbExt}`, thumbBlob);
224 } catch (error) {
225 console.error('Error downloading thumbnail:', error);
226 }
227 }
228
229 // Wait for all downloads to complete
230 await Promise.all(texturePromises);
231
232 // Add README file
233 zip.file("README.txt",
234 `Material: ${material.name}\n` +
235 `Resolution: ${resolution}\n` +
236 `Source: https://polyhaven.com/a/${material.id}\n` +
237 `Downloaded: ${new Date().toISOString()}\n\n` +
238 `Contains the following files:\n` +
239 `- ${material.id}.mtlx\n` +
240 Object.keys(textureFiles).map(path => `- ${path}`).join('\n') +
241 (material.thumb_url ? `\n- ${material.id}_thumbnail.png` : '')
242 );
243
244 // Generate the ZIP file
245 return await zip.generateAsync({ type: 'blob' });
246
247 } catch (error) {
248 console.error('Error creating MaterialX package:', error);
249 throw error;
250 }
251 }
